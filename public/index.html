const catalogEntries = []; const GN_MATH_ZONES = "https://cdn.jsdelivr.net/gh/gn-math/assets@main/zones.json"; const GN_MATH_HTML = "https://cdn.jsdelivr.net/gh/gn-math/html@main"; const GN_MATH_COVERS = "https://cdn.jsdelivr.net/gh/gn-math/covers@main"; function resolveGnMathUrl(value, base) { if (!value || typeof value !== "string") return ""; if (value.startsWith("http://") || value.startsWith("https://")) { return value; } value = value .replaceAll("{HTML_URL}", GN_MATH_HTML) .replaceAll("{COVER_URL}", GN_MATH_COVERS); if (value.startsWith("local")) return ""; if (value.startsWith("/")) { value = value.slice(1); } if (!value.startsWith("http")) { value = `${base}/${value}`; } return value; } async function loadGnMathGames() { try { const response = await fetch(GN_MATH_ZONES); if (!response.ok) { throw new Error("Could not load the GN-Math game list."); } const zones = await response.json(); const games = zones .filter(game => game && game.name && game.url) .map(game => { const url = resolveGnMathUrl( game.url, GN_MATH_HTML ); const image = resolveGnMathUrl( game.cover || "", GN_MATH_COVERS ); return { id: `gn-math-${game.id}`, title: game.name, description: `Play ${game.name}`, category: "Arcade", image, url, catalog: true, createdAt: Number(game.id) || Date.now() }; }) .filter(game => game.url); // Avoid adding duplicates if the loader runs again. const existingIds = new Set( catalogEntries.map(game => game.id) ); for (const game of games) { if (!existingIds.has(game.id)) { catalogEntries.push(game); } } console.log( `HauntHub loaded ${games.length} GN-Math game entries.` ); // Refresh the existing HauntHub displays, if available. if (typeof renderGames === "function") { renderGames(); } if (typeof renderHome === "function") { renderHome(); } if (typeof renderRecentGames === "function") { renderRecentGames(); } } catch (error) { console.error("GN-Math loading error:", error); } } loadGnMathGames();
